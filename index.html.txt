<script>
  (function(){
    var yesBtn = document.getElementById('yes-btn');
    var noBtn = document.getElementById('no-btn');
    var modal = document.getElementById('modal-overlay');
    var modalOk = document.getElementById('modal-ok');
    var confettiLayer = document.getElementById('confetti-layer');

    var confettiColors = ['#D6547A', '#FFC9DA', '#F6D976', '#F4A6BE', '#B23A5E', '#FFFFFF'];

    function launchConfetti(){
      var count = 120;
      for (var i = 0; i < count; i++){
        var piece = document.createElement('div');
        piece.className = 'confetti-piece';

        var size = 6 + Math.random() * 8;
        var isRound = Math.random() > 0.5;
        piece.style.width = size + 'px';
        piece.style.height = (isRound ? size : size * 1.6) + 'px';
        piece.style.background = confettiColors[Math.floor(Math.random() * confettiColors.length)];
        if (isRound) piece.style.borderRadius = '50%';

        var left = Math.random() * 100;
        piece.style.left = left + 'vw';

        var duration = 2.6 + Math.random() * 1.8;
        var delay = Math.random() * 0.4;
        piece.style.animationDuration = duration + 's';
        piece.style.animationDelay = delay + 's';

        confettiLayer.appendChild(piece);

        (function(el, totalTime){
          setTimeout(function(){
            if (el.parentNode) el.parentNode.removeChild(el);
          }, totalTime * 1000);
        })(piece, duration + delay + 0.2);
      }
    }

    yesBtn.addEventListener('click', function(){
      launchConfetti();
    });
    window.launchConfetti = launchConfetti;

    function openModal(){
      modal.classList.add('open');
    }
    function closeModal(){
      modal.classList.remove('open');
    }
    window.openDateModal = openModal;

    noBtn.addEventListener('click', openModal);
    modalOk.addEventListener('click', closeModal);
    modal.addEventListener('click', function(e){
      if (e.target === modal) closeModal();
    });
    document.addEventListener('keydown', function(e){
      if (e.key === 'Escape') closeModal();
    });
  })();
</script>
